local AutoDamage = {
    Connection = nil,
    LastDamage = 0,
    DamageAmount = 999999
}

function AutoDamage:FindDamageRemote()
    -- Tìm remote damage hợp lệ trong ReplicatedStorage
    for _, obj in pairs(ReplicatedStorage:GetDescendants()) do
        if obj:IsA("RemoteEvent") or obj:IsA("RemoteFunction") then
            local name = obj.Name:lower()
            if name:find("damage") or name:find("hit") or name:find("attack") or name:find("combat") or name:find("deal") then
                return obj
            end
        end
    end
    
    -- Tìm trong workspace
    for _, obj in pairs(workspace:GetDescendants()) do
        if obj:IsA("RemoteEvent") or obj:IsA("RemoteFunction") then
            local name = obj.Name:lower()
            if name:find("damage") or name:find("hit") or name:find("attack") or name:find("combat") then
                return obj
            end
        end
    end
    
    -- Tìm trong LocalPlayer
    local character = LocalPlayer.Character
    if character then
        for _, obj in pairs(character:GetDescendants()) do
            if obj:IsA("RemoteEvent") or obj:IsA("RemoteFunction") then
                local name = obj.Name:lower()
                if name:find("damage") or name:find("hit") or name:find("attack") then
                    return obj
                end
            end
        end
    end
    
    return nil
end

function AutoDamage:DamageEnemy(enemy)
    local character = LocalPlayer.Character
    if not character or not character:FindFirstChild("HumanoidRootPart") then return false end
    
    -- Cách 1: Dùng remote damage hợp lệ
    local damageRemote = self:FindDamageRemote()
    if damageRemote then
        if damageRemote:IsA("RemoteEvent") then
            damageRemote:FireServer(enemy, self.DamageAmount)
            return true
        elseif damageRemote:IsA("RemoteFunction") then
            damageRemote:InvokeServer(enemy, self.DamageAmount)
            return true
        end
    end
    
    -- Cách 2: Dùng tool đánh quái
    for _, tool in pairs(character:GetChildren()) do
        if tool:IsA("Tool") and tool:FindFirstChild("Handle") then
            -- Dịch chuyển tool đến quái
            tool.Handle.CFrame = enemy.HumanoidRootPart.CFrame
            tool:Activate()
            task.wait(0.01)
            tool:Deactivate()
            return true
        end
    end
    
    -- Cách 3: Trang bị tool từ backpack rồi đánh
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if humanoid then
        for _, tool in pairs(LocalPlayer.Backpack:GetChildren()) do
            if tool:IsA("Tool") and tool:FindFirstChild("Handle") then
                humanoid:EquipTool(tool)
                task.wait(0.01)
                tool:Activate()
                task.wait(0.01)
                tool:Deactivate()
                return true
            end
        end
    end
    
    -- Cách 4: Dùng RemoteEvent tấn công thông thường
    local attackRemotes = {"MeleeAttack", "Attack", "Click", "Hit", "Swing", "Punch", "Combat"}
    for _, remoteName in pairs(attackRemotes) do
        local remote = character:FindFirstChild(remoteName) or LocalPlayer.Backpack:FindFirstChild(remoteName)
        if remote and remote:IsA("RemoteEvent") then
            remote:FireServer(enemy)
            return true
        end
    end
    
    return false
end

function AutoDamage:Start()
    if self.Connection then return end
    
    self.Connection = RunService.RenderStepped:Connect(function()
        if not Settings.AutoDamage then return end
        
        local character = LocalPlayer.Character
        if not character or not character:FindFirstChild("HumanoidRootPart") then return end
        
        local rootPart = character.HumanoidRootPart
        local currentTime = os.clock()
        
        -- Kiểm tra delay
        if currentTime - self.LastDamage < Settings.AttackDelay then return end
        
        -- Tìm tất cả quái trong phạm vi
        local damagedEnemy = false
        
        -- Tìm trong workspace.Enemies
        for _, enemy in pairs(workspace.Enemies:GetChildren()) do
            if enemy:FindFirstChild("HumanoidRootPart") and enemy:FindFirstChild("Humanoid") then
                if enemy.Humanoid.Health > 0 then
                    local distance = (rootPart.Position - enemy.HumanoidRootPart.Position).Magnitude
                    
                    if distance <= Settings.DamageRange then
                        if self:DamageEnemy(enemy) then
                            self.LastDamage = currentTime
                            damagedEnemy = true
                            break
                        end
                    end
                end
            end
        end
        
        -- Tìm thêm trong workspace
        if not damagedEnemy then
            for _, obj in pairs(workspace:GetDescendants()) do
                if obj:IsA("Model") and obj:FindFirstChild("Humanoid") and obj:FindFirstChild("HumanoidRootPart") then
                    if obj.Humanoid.Health > 0 and obj ~= LocalPlayer.Character then
                        local isPlayer = false
                        for _, player in pairs(Players:GetPlayers()) do
                            if player.Character == obj then
                                isPlayer = true
                                break
                            end
                        end
                        
                        if not isPlayer then
                            local distance = (rootPart.Position - obj.HumanoidRootPart.Position).Magnitude
                            
                            if distance <= Settings.DamageRange then
                                if self:DamageEnemy(obj) then
                                    self.LastDamage = currentTime
                                    break
                                end
                            end
                        end
                    end
                end
            end
        end
    end)
end

function AutoDamage:Stop()
    if self.Connection then
        self.Connection:Disconnect()
        self.Connection = nil
    end
end

-- AUTO FARM - TỰ DỊCH CHUYỂN
-- ============================================
local AutoFarm = {
    Connection = nil,
    CurrentTarget = nil
}

function AutoFarm:FindNearestEnemy()
    local character = LocalPlayer.Character
    if not character or not character:FindFirstChild("HumanoidRootPart") then return nil end
    
    local rootPart = character.HumanoidRootPart
    local nearestEnemy = nil
    local nearestDistance = Settings.AttackRange
    
    for _, enemy in pairs(workspace.Enemies:GetChildren()) do
        if enemy:FindFirstChild("HumanoidRootPart") and enemy:FindFirstChild("Humanoid") then
            if enemy.Humanoid.Health > 0 then
                local distance = (rootPart.Position - enemy.HumanoidRootPart.Position).Magnitude
                if distance < nearestDistance then
                    nearestDistance = distance
                    nearestEnemy = enemy
                end
            end
        end
    end
    
    if not nearestEnemy then
        for _, obj in pairs(workspace:GetDescendants()) do
            if obj:IsA("Model") and obj:FindFirstChild("Humanoid") and obj:FindFirstChild("HumanoidRootPart") then
                if obj.Humanoid.Health > 0 and obj ~= LocalPlayer.Character then
                    local isPlayer = false
                    for _, player in pairs(Players:GetPlayers()) do
                        if player.Character == obj then
                            isPlayer = true
                            break
                        end
                    end
                    
                    if not isPlayer then
                        local distance = (rootPart.Position - obj.HumanoidRootPart.Position).Magnitude
                        if distance < nearestDistance then
                            nearestDistance = distance
                            nearestEnemy = obj
                        end
                    end
                end
            end
        end
    end
    
    return nearestEnemy
end

function AutoFarm:Start()
    if self.Connection then return end
    
    self.Connection = RunService.RenderStepped:Connect(function()
        if not Settings.AutoFarm then return end
        
        local character = LocalPlayer.Character
        if not character or not character:FindFirstChild("HumanoidRootPart") then return end
        
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        if not humanoid or humanoid.Health <= 0 then return end
        
        local rootPart = character.HumanoidRootPart
        
        local enemy = self:FindNearestEnemy()
        
        if enemy then
            self.CurrentTarget = enemy
            rootPart.CFrame = CFrame.new(enemy.HumanoidRootPart.Position + Vector3.new(0, 3, 2))
            rootPart.CFrame = CFrame.lookAt(rootPart.Position, enemy.HumanoidRootPart.Position)
        end
    end)
end

function AutoFarm:Stop()
    if self.Connection then
        self.Connection:Disconnect()
        self.Connection = nil
    end
    self.CurrentTarget = nil
end

-- FIX LAG
-- ============================================
local FixLag = {
    Enabled = false,
    OriginalGraphics = {}
}

function FixLag:Enable()
    if self.Enabled then return end
    self.Enabled = true
    
    self.OriginalGraphics = {
        GlobalShadows = Lighting.GlobalShadows,
        Brightness = Lighting.Brightness,
        FogEnd = Lighting.FogEnd,
        FogStart = Lighting.FogStart
    }
    
    Lighting.GlobalShadows = false
    Lighting.Brightness = 1.5
    Lighting.FogEnd = 100
    Lighting.FogStart = 50
    
    task.spawn(function()
        while self.Enabled do
            for _, obj in pairs(workspace:GetDescendants()) do
                if obj:IsA("Decal") or obj:IsA("ParticleEmitter") or obj:IsA("Smoke") or obj:IsA("Fire") or obj:IsA("Texture") then
                    obj:Destroy()
                end
            end
            task.wait(5)
        end
    end)
    
    Notify("BIN BELL", "Fix Lag đã bật!")
end

function FixLag:Disable()
    if not self.Enabled then return end
    self.Enabled = false
    
    if self.OriginalGraphics.GlobalShadows ~= nil then
        Lighting.GlobalShadows = self.OriginalGraphics.GlobalShadows
        Lighting.Brightness = self.OriginalGraphics.Brightness
        Lighting.FogEnd = self.OriginalGraphics.FogEnd
        Lighting.FogStart = self.OriginalGraphics.FogStart
    end
    
    Notify("BIN BELL", "Fix Lag đã tắt!")
end

local function AddHighlight(target, color)
    if not target or not target:IsA("Model") then return end
    if target:FindFirstChild("BinBellHighlight") then return end
    
    local highlight = Instance.new("Highlight")
    highlight.Name = "BinBellHighlight"
    highlight.FillColor = color
    highlight.OutlineColor = color
    highlight.FillTransparency = 0.5
    highlight.OutlineTransparency = 0
    highlight.Parent = target
    
    table.insert(ESPHighlights, highlight)
end

local function ClearHighlights()
    for _, h in pairs(ESPHighlights) do
        if h and h.Parent then
            h:Destroy()
        end
    end
    ESPHighlights = {}
end

local function StartESP()
    if Connections.ESP then return end
    
    Connections.ESP = RunService.RenderStepped:Connect(function()
        if Settings.ESPEnemies then
            for _, enemy in pairs(workspace.Enemies:GetChildren()) do
                if enemy:FindFirstChild("Humanoid") and enemy:FindFirstChild("HumanoidRootPart") then
                    if enemy.Humanoid.Health > 0 then
                        if not enemy:FindFirstChild("BinBellHighlight") then
                            AddHighlight(enemy, Color3.fromRGB(255, 0, 0))
                        end
                    end
                end
            end
        end
        
        if Settings.ESPPlayers then
            for _, player in pairs(Players:GetPlayers()) do
                if player ~= LocalPlayer and player.Character then
                    local char = player.Character
                    if char:FindFirstChild("Humanoid") and char:FindFirstChild("HumanoidRootPart") then
                        if char.Humanoid.Health > 0 then
                            if not char:FindFirstChild("BinBellHighlight") then
                                AddHighlight(char, Color3.fromRGB(0, 255, 255))
                            end
                        end
                    end
                end
            end
        end
    end)
end

function StopESP()
    ClearHighlights()
    if Connections.ESP then
        Connections.ESP:Disconnect()
        Connections.ESP = nil
    end
end

    fill.BackgroundColor3 = Color3.fromRGB(255, 0, 0)
    fill.BorderSizePixel = 0
    fill.Parent = sliderBg
    
    local fillCorner = Instance.new("UICorner")
    fillCorner.CornerRadius = UDim.new(1, 0)
    fillCorner.Parent = fill
    
    local knob = Instance.new("Frame")
    knob.Size = UDim2.new(0, 14, 0, 14)
    knob.Position = UDim2.new((default - min) / (max - min), -7, 0, -4)
    knob.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    knob.BorderSizePixel = 0
    knob.Parent = sliderBg
    
    local knobCorner = Instance.new("UICorner")
    knobCorner.CornerRadius = UDim.new(1, 0)
    knobCorner.Parent = knob
    
    local dragging = false
    
    local function updateFromInput(input)
        local relativeX = input.Position.X - sliderBg.AbsolutePosition.X
        local percent = math.clamp(relativeX / sliderBg.AbsoluteSize.X, 0, 1)
        local value = min + (max - min) * percent
        
        fill.Size = UDim2.new(percent, 0, 1, 0)
        knob.Position = UDim2.new(percent, -7, 0, -4)
        label.Text = name .. ": " .. string.format("%.0f", value)
        
        if callback then
            callback(value)
        end
    end
    
    sliderBg.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            updateFromInput(input)
        end
    end)
    
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            updateFromInput(input)
        end
    end)
    
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
end
