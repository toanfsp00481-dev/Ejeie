local AutoFarm = {
    Enabled = false,
    CurrentTarget = nil,
    AttackRange = 15,
    LastAttack = 0,
    AttackDelay = 0.3
}

function AutoFarm:Start()
    self.Enabled = true
    
    self.Connection = RunService.RenderStepped:Connect(function()
        if not self.Enabled then return end
        
        local character = LocalPlayer.Character
        if not character or not character:FindFirstChild("HumanoidRootPart") then return end
        
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        if not humanoid or humanoid.Health <= 0 then return end
        
        -- Tìm enemy gần nhất
        local nearestEnemy = nil
        local nearestDistance = self.AttackRange
        
        for _, enemy in pairs(Workspace.Enemies:GetChildren()) do
            if enemy:FindFirstChild("HumanoidRootPart") and enemy:FindFirstChild("Humanoid") then
                local enemyHumanoid = enemy.Humanoid
                if enemyHumanoid.Health > 0 then
                    local distance = (character.HumanoidRootPart.Position - enemy.HumanoidRootPart.Position).Magnitude
                    if distance < nearestDistance then
                        nearestDistance = distance
                        nearestEnemy = enemy
                    end
                end
            end
        end
        
        if nearestEnemy then
            self.CurrentTarget = nearestEnemy
            
            -- Dịch chuyển đến enemy
            local targetPos = nearestEnemy.HumanoidRootPart.Position
            character.HumanoidRootPart.CFrame = CFrame.new(targetPos + Vector3.new(0, 3, 0))
            
            -- Tấn công
            if os.clock() - self.LastAttack >= self.AttackDelay then
                self.LastAttack = os.clock()
                
                -- Thử nhiều remote khác nhau
                local args = {
                    [1] = "Attack",
                    [2] = nearestEnemy
                }
                
                for _, remoteName in pairs({"MeleeAttack", "Attack", "Click", "Hit"}) do
                    local remote = LocalPlayer.Backpack:FindFirstChild(remoteName) or 
                                   character:FindFirstChild(remoteName)
                    if remote and remote:IsA("RemoteEvent") then
                        remote:FireServer(unpack(args))
                        break
                    end
                end
            end
        end
    end)
end

function AutoFarm:Stop()
    self.Enabled = false
    if self.Connection then
        self.Connection:Disconnect()
        self.Connection = nil
    end
end

-- HỆ THỐNG CHỐNG AFK
-- ============================================
local AntiAFK = {
    Enabled = false,
    Connection = nil
}

function AntiAFK:Start()
    self.Enabled = true
    
    self.Connection = RunService.RenderStepped:Connect(function()
        if not self.Enabled then return end
        
        -- Gửi input giả để chống AFK
        VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.LeftShift, false, nil)
        VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.LeftShift, false, nil)
    end)
end

function AntiAFK:Stop()
    self.Enabled = false
    if self.Connection then
        self.Connection:Disconnect()
        self.Connection = nil
    end
end
