--/ Axiom PlayerGui Direct Context Injection /--
local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local ReplicatedStorage = game:GetService("ReplicatedStorage")

local LocalPlayer = Players.LocalPlayer
-- Đảm bảo luồng giao diện tìm đúng phân cấp PlayerGui mà game bắt buộc hiển thị
local TargetGuiParent = LocalPlayer:WaitForChild("PlayerGui")

local AxiomState = {
    AutoFarm = false,
    NoClip = false,
    ESP = false,
    TargetEnemy = "Monkey",
    YOffset = 8.5
}

local AxiomFixedMenu = {
    Connections = {},
    VisualCache = {}
}

-- Xóa sạch thực thể trùng lặp cũ để giải phóng không gian tên miền
if TargetGuiParent:FindFirstChild("AxiomDirectFallback") then
    TargetGuiParent.AxiomDirectFallback:Destroy()
end

function AxiomFixedMenu:ToggleNoClip(state)
    AxiomState.NoClip = state
    if self.Connections["NoClip"] then self.Connections["NoClip"]:Disconnect() self.Connections["NoClip"] = nil end
    if state then
        self.Connections["NoClip"] = RunService.Stepped:Connect(function()
            if LocalPlayer.Character then
                for _, Part in ipairs(LocalPlayer.Character:GetChildren()) do
                    if Part:IsA("BasePart") then Part.CanCollide = false end
                end
            end
        end)
    end
end

function AxiomFixedMenu:ToggleESP(state)
    AxiomState.ESP = state
    for Enemy, Highlight in pairs(self.VisualCache) do
        if Highlight then Highlight:Destroy() end
        self.VisualCache[Enemy] = nil
    end
    if self.Connections["ESP"] then self.Connections["ESP"]:Disconnect() self.Connections["ESP"] = nil end
    
    if state then
        self.Connections["ESP"] = RunService.RenderStepped:Connect(function()
            local Enemies = workspace:FindFirstChild("Enemies") or workspace:FindFirstChild("Monsters")
            if not Enemies then return end
            for _, Enemy in ipairs(Enemies:GetChildren()) do
                if Enemy:IsA("Model") and Enemy:FindFirstChild("HumanoidRootPart") and Enemy:FindFirstChild("Humanoid") and Enemy.Humanoid.Health > 0 then
                    if not self.VisualCache[Enemy] then
                        local Box = Instance.new("Highlight")
                        Box.Name = "AxiomESP"
                        Box.FillColor = Color3.fromRGB(0, 255, 120)
                        Box.FillTransparency = 0.6
                        Box.Adornee = Enemy
                        Box.Parent = TargetGuiParent
                        self.VisualCache[Enemy] = Box
                    end
                else
                    if self.VisualCache[Enemy] then self.VisualCache[Enemy]:Destroy() self.VisualCache[Enemy] = nil end
                end
            end
        end)
    end
end

function AxiomFixedMenu:ToggleAutoFarm(state)
    AxiomState.AutoFarm = state
    if self.Connections["AutoFarm"] then self.Connections["AutoFarm"]:Disconnect() self.Connections["AutoFarm"] = nil end
    
    if state then
        local CurrentTarget = nil
        self.Connections["AutoFarm"] = RunService.Heartbeat:Connect(function()
            local Character = LocalPlayer.Character
            local Root = Character and Character:FindFirstChild("HumanoidRootPart")
            if not Root then return end
            
            local BV = Root:FindFirstChild("AxiomHold") or Instance.new("BodyVelocity")
            if not BV.Parent then
                BV.Name = "AxiomHold"
                BV.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
                BV.Velocity = Vector3.new(0,0,0)
                BV.Parent = Root
            end
            
            local Enemies = workspace:FindFirstChild("Enemies") or workspace:FindFirstChild("Monsters")
            if Enemies and (not CurrentTarget or not CurrentTarget.Parent or CurrentTarget.Humanoid.Health <= 0) then
                local MaxDist = math.huge
                for _, E in ipairs(Enemies:GetChildren()) do
                    if E:IsA("Model") and E:FindFirstChild("HumanoidRootPart") and E:FindFirstChild("Humanoid") and E.Humanoid.Health > 0 then
                        local Dist = (Root.Position - E.HumanoidRootPart.Position).Magnitude
                        if Dist < MaxDist then MaxDist = Dist CurrentTarget = E end
                    end
                end
            end
            
            if CurrentTarget and CurrentTarget:FindFirstChild("HumanoidRootPart") then
                Root.CFrame = CurrentTarget.HumanoidRootPart.CFrame * CFrame.new(0, AxiomState.YOffset, 0)
                local Remote = ReplicatedStorage:FindFirstChild("Remotes") and ReplicatedStorage.Remotes:FindFirstChild("Validator")
                if Remote then Remote:FireServer("AttackContinuous", CurrentTarget.HumanoidRootPart.Position) end
            end
        end)
    else
        local Root = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
        if Root and Root:FindFirstChild("AxiomHold") then Root.AxiomHold:Destroy() end
    end
end

function AxiomFixedMenu:InitializeRenderPipeline()
    local ScreenGui = Instance.new("ScreenGui")
    ScreenGui.Name = "AxiomDirectFallback"
    ScreenGui.ResetOnSpawn = false
    ScreenGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
    
    local Frame = Instance.new("Frame")
    Frame.Size = UDim2.new(0, 260, 0, 240)
    Frame.Position = UDim2.new(0.3, 0, 0.3, 0)
    Frame.BackgroundColor3 = Color3.fromRGB(15, 15, 18)
    Frame.BorderSizePixel = 1
    Frame.Active = true
    Frame.Draggable = true
    Frame.Parent = ScreenGui
    
    local Banner = Instance.new("TextLabel")
    Banner.Size = UDim2.new(1, 0, 0, 35)
    Banner.BackgroundColor3 = Color3.fromRGB(25, 25, 30)
    Banner.Text = "AXIOM CORE RENDERER"
    Banner.TextColor3 = Color3.fromRGB(255, 255, 255)
    Banner.Font = Enum.Font.Code
    Banner.TextSize = 13
    Banner.Parent = Frame
    
    local function AppendButton(text, offset, trigger)
        local Button = Instance.new("TextButton")
        Button.Size = UDim2.new(0.9, 0, 0, 40)
        Button.Position = UDim2.new(0.05, 0, 0, offset)
        Button.BackgroundColor3 = Color3.fromRGB(35, 35, 40)
        Button.Text = text .. " : STABLE"
        Button.TextColor3 = Color3.fromRGB(255, 255, 255)
        Button.Font = Enum.Font.SourceSans
        Button.TextSize = 14
        
        local active = false
        Button.MouseButton1Click:Connect(function()
            active = not active
            Button.BackgroundColor3 = active and Color3.fromRGB(0, 200, 100) or Color3.fromRGB(35, 35, 40)
            trigger(active)
        end)
        Button.Parent = Frame
    end
    
    AppendButton("Tự động cày (Auto Farm)", 55, function(s) self:ToggleAutoFarm(s) end)
    AppendButton("Định vị thực thể (ESP Radar)", 110, function(s) self:ToggleESP(s) end)
    AppendButton("Xuyên tường vật lý (NoClip)", 165, function(s) self:ToggleNoClip(s) end)
    
    ScreenGui.Parent = TargetGuiParent
end

AxiomFixedMenu:InitializeRenderPipeline()
