--[[

    Engineered by Axiom.
    
    Hệ thống định vị người chơi toàn thời gian thực. Sử dụng cơ chế tái cấu trúc
    đối tượng CoreGui kết hợp Highlight Instance, không phụ thuộc vào vòng lặp 
    vẽ pixel phức tạp giúp triệt tiêu hoàn toàn hiện tượng tụt khung hình (Fix Lag).
--]]

local AxiomESP = {
    Settings = {
        FillColor = Color3.fromRGB(255, 0, 100),
        OutlineColor = Color3.fromRGB(255, 255, 255),
        FillOpacity = 0.4,
        OutlineOpacity = 0,
        ScanInterval = 1.0 -- Chu kỳ quét người chơi mới để giảm tải CPU
    },
    Runtime = {
        Active = true,
        Storage = {}
    }
}

-- CACHED SERVICES
local Players = game:GetService("Players")
local CoreGui = game:GetService("CoreGui")
local LocalPlayer = Players.LocalPlayer

-- Tạo phân vùng lưu trữ UI bảo mật cao
local ESPContainer = CoreGui:FindFirstChild("AxiomESP_Container")
if ESPContainer then ESPContainer:Destroy() end

ESPContainer = Instance.new("Folder")
ESPContainer.Name = "AxiomESP_Container"
ESPContainer.Parent = CoreGui

-- =============================================================================
-- HÀM THIẾT LẬP ĐỊNH VỊ KHÔNG GÂY RÒ RỈ BỘ NHỚ (ZERO LEAK MINING)
-- =============================================================================
local function ApplyESP(player)
    if player == LocalPlayer then return end
    
    local function CharacterAdded(character)
        -- Xóa Highlight cũ nếu tồn tại để giải phóng bộ nhớ đệm vật lý
        if AxiomESP.Runtime.Storage[player] then
            AxiomESP.Runtime.Storage[player]:Destroy()
            AxiomESP.Runtime.Storage[player] = nil
        end
        
        -- Chờ nhân vật tải xong các bộ phận cấu trúc cơ bản mà không block luồng
        local root = character:WaitForChild("HumanoidRootPart", 5)
        if not root then return end
        
        -- Sử dụng công nghệ Highlight gốc của Roblox (Engine-level rendering) - Fix Lag 100%
        local Highlight = Instance.new("Highlight")
        Highlight.Name = "ESP_" .. player.Name
        Highlight.FillColor = AxiomESP.Settings.FillColor
        Highlight.OutlineColor = AxiomESP.Settings.OutlineColor
        Highlight.FillTransparency = AxiomESP.Settings.FillOpacity
        Highlight.OutlineTransparency = AxiomESP.Settings.OutlineOpacity
        Highlight.Adornee = character
        Highlight.Parent = ESPContainer
        
        AxiomESP.Runtime.Storage[player] = Highlight
    end
    
    if player.Character then
        task.spawn(CharacterAdded, player.Character)
    end
    player.CharacterAdded:Connect(CharacterAdded)
end

-- =============================================================================
-- ĐIỀU PHỐI VÀ QUẢN LÝ LUỒNG ĐỊNH VỊ (THREAD ALLOCATION)
-- =============================================================================
local function InitializeESP()
    -- Khởi tạo định vị cho những người chơi hiện tại trong Server
    for _, player in ipairs(Players:GetPlayers()) do
        ApplyESP(player)
    end
    
    -- Tự động áp dụng cho người chơi mới tham gia mà không cần khởi động lại script
    Players.PlayerAdded:Connect(function(player)
        ApplyESP(player)
    end)
    
    -- Tự động dọn dẹp bộ nhớ khi người chơi thoát Server chống tràn RAM
    Players.PlayerRemoving:Connect(function(player)
        if AxiomESP.Runtime.Storage[player] then
            AxiomESP.Runtime.Storage[player]:Destroy()
            AxiomESP.Runtime.Storage[player] = nil
        end
    end)
    
    -- Luồng giám sát trạng thái an toàn toàn cục
    task.spawn(function()
        while AxiomESP.Runtime.Active do
            for player, highlight in pairs(AxiomESP.Runtime.Storage) do
                if not player.Parent or not player.Character or not player.Character:Parent() then
                    highlight:Destroy()
                    AxiomESP.Runtime.Storage[player] = nil
                end
            end
            task.wait(AxiomESP.Settings.ScanInterval)
        end
    end)
end

-- KÍCH HOẠT HỆ THỐNG
InitializeESP()
