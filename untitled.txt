-- ========== LOGIC ==========
local noclipConns = {}
function ApplyNoclip()
    for _, c in pairs(noclipConns) do c:Disconnect() end
    noclipConns = {}
    local char = LocalPlayer.Character
    if not char then return end
    local conn = RunService.Stepped:Connect(function()
        if not Config.Noclip or not LocalPlayer.Character then return end
        for _, part in pairs(LocalPlayer.Character:GetDescendants()) do
            if part:IsA("BasePart") and part.CanCollide then
                part.CanCollide = false
            end
        end
    end)
    table.insert(noclipConns, conn)
end

LocalPlayer.CharacterAdded:Connect(function()
    task.wait(1)
    if Config.Noclip then ApplyNoclip() end
end)

local wallbangConns = {}
local originalSizes = {}
function ApplyWallbang()
    if not Config.Wallbang then return end
    RemoveWallbang()
    local conn = RunService.Heartbeat:Connect(function()
        if not Config.Wallbang then return end
        for _, p in pairs(Players:GetPlayers()) do
            if p ~= LocalPlayer and p.Character then
                for _, part in pairs(p.Character:GetDescendants()) do
                    if part:IsA("BasePart") then
                        if not originalSizes[part] then originalSizes[part] = part.Size end
                        if part.Name == "Head" then
                            part.Size = Vector3.new(2.5, 2.5, 2.5)
                        else
                            part.Size = originalSizes[part] * 1.3
                        end
                        part.CanCollide = false
                        part.Transparency = math.max(part.Transparency, 0.85)
                    end
                end
            end
        end
    end)
    table.insert(wallbangConns, conn)
end

function RemoveWallbang()
    for _, c in pairs(wallbangConns) do c:Disconnect() end
    wallbangConns = {}
    for part, size in pairs(originalSizes) do
        if part and part.Parent then
            part.Size = size
            part.CanCollide = true
        end
    end
    originalSizes = {}
end

-- AIM
local function IsValidTarget(player)
    if player == LocalPlayer then return false end
    if not player.Character then return false end
    local hum = player.Character:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return false end
    if not player.Character:FindFirstChild("Head") then return false end
    if Config.TeamCheck and player.Team and LocalPlayer.Team and player.Team == LocalPlayer.Team then return false end
    if Config.WallCheck then
        local ray = Ray.new(Camera.CFrame.Position, (player.Character.Head.Position - Camera.CFrame.Position).Unit * 500)
        local hit = Workspace:FindPartOnRay(ray, LocalPlayer.Character)
        if hit and not hit:IsDescendantOf(player.Character) then return false end
    end
    return true
end

local function GetClosestHead()
    local closest, shortest = nil, Config.AimFOV
    for _, player in pairs(Players:GetPlayers()) do
        if IsValidTarget(player) then
            local head = player.Character.Head
            local screenPos, onScreen = Camera:WorldToViewportPoint(head.Position)
            if onScreen then
                local dist = (Vector2.new(screenPos.X, screenPos.Y) - Vector2.new(Mouse.X, Mouse.Y)).Magnitude
                if dist < shortest then
                    shortest = dist
                    closest = head
                end
            end
        end
    end
    return closest
end

local function AimAt(target, smoothness)
    if not target then return end
    local cur = Camera.CFrame
    local new = CFrame.new(cur.Position, target.Position)
    Camera.CFrame = cur:Lerp(new, smoothness)
end

local function RunAim()
    if not UIS:IsMouseButtonPressed(Config.AimKey) then return end
    if Config.HeadAimbot then
        local head = GetClosestHead()
        if head then AimAt(head, Config.HeadSmoothness) end
    elseif Config.AimAssist then
        local target = GetClosestHead()
        if target then AimAt(target, Config.AimSmoothness) end
    end
end

local function UpdateESP()
    for _, player in pairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local ex = player.Character:FindFirstChild("RivalsESP")
            if Config.ESP and not ex then
                local H = Instance.new("Highlight")
                H.Name = "RivalsESP"
                H.FillColor = Theme.Red
                H.FillTransparency = 0.55
                H.OutlineColor = Color3.fromRGB(255, 255, 255)
                H.OutlineTransparency = 0
                H.Parent = player.Character
            elseif not Config.ESP and ex then
                ex:Destroy()
            end
        end
    end
end

RunService.RenderStepped:Connect(function()
    UpdateESP()
    RunAim()
end)

Players.PlayerAdded:Connect(function(p)
    p.CharacterAdded:Connect(function()
        task.wait(1)
        UpdateESP()
    end)
end)
