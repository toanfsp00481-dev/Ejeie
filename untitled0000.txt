-- Kiểm tra người chơi
function IsPlayer(model)
    for _, player in pairs(Players:GetPlayers()) do
        if player.Character == model then
            return true
        end
    end
    return false
end

-- Tìm quái
function FindEnemies()
    local enemies = {}
    
    if workspace:FindFirstChild("Enemies") then
        for _, enemy in pairs(workspace.Enemies:GetChildren()) do
            if enemy:FindFirstChild("HumanoidRootPart") and enemy:FindFirstChild("Humanoid") then
                if enemy.Humanoid.Health > 0 then
                    table.insert(enemies, enemy)
                end
            end
        end
    end
    
    for _, obj in pairs(workspace:GetDescendants()) do
        if obj:IsA("Model") and obj:FindFirstChild("Humanoid") and obj:FindFirstChild("HumanoidRootPart") then
            if obj.Humanoid.Health > 0 and obj ~= LocalPlayer.Character and not IsPlayer(obj) then
                local alreadyAdded = false
                for _, existing in pairs(enemies) do
                    if existing == obj then
                        alreadyAdded = true
                        break
                    end
                end
                
                if not alreadyAdded then
                    table.insert(enemies, obj)
                end
            end
        end
    end
    
    return enemies
end

-- Tìm quái gần nhất
function FindNearestEnemy(range)
    local character = LocalPlayer.Character
    if not character or not character:FindFirstChild("HumanoidRootPart") then return nil end
    
    local rootPart = character.HumanoidRootPart
    local allEnemies = FindEnemies()
    local nearest = nil
    local nearestDistance = range
    
    for _, enemy in pairs(allEnemies) do
        local distance = (rootPart.Position - enemy.HumanoidRootPart.Position).Magnitude
        if distance < nearestDistance then
            nearestDistance = distance
            nearest = enemy
        end
    end
    
    return nearest
end

-- Kiểm tra vũ khí đang trang bị
function GetEquippedWeapon()
    local character = LocalPlayer.Character
    if not character then return nil end
    
    -- Tìm tool đang cầm
    for _, tool in pairs(character:GetChildren()) do
        if tool:IsA("Tool") and tool:FindFirstChild("Handle") then
            return tool
        end
    end
    
    return nil
end

-- Kiểm tra vũ khí trong backpack
function GetBackpackWeapon()
    for _, tool in pairs(LocalPlayer.Backpack:GetChildren()) do
        if tool:IsA("Tool") and tool:FindFirstChild("Handle") then
            return tool
        end
    end
    return nil
end

-- Trang bị vũ khí
function EquipWeapon(tool)
    local character = LocalPlayer.Character
    if not character or not character:FindFirstChildOfClass("Humanoid") then return false end
    
    character.Humanoid:EquipTool(tool)
    return true
end

-- Đánh bằng vũ khí
function AttackWithWeapon(enemy)
    local character = LocalPlayer.Character
    if not character then return false end
    
    local weapon = GetEquippedWeapon()
    
    -- Nếu chưa có vũ khí, trang bị từ backpack
    if not weapon then
        weapon = GetBackpackWeapon()
        if weapon then
            EquipWeapon(weapon)
            task.wait(0.1)
            weapon = GetEquippedWeapon()
        end
    end
    
    if not weapon then
        Notify("BIN BELL", "Chua co vu khi! Hay trang bi nam dam hoac kiem")
        return false
    end
    
    -- Kiểm tra loại vũ khí
    local weaponName = weapon.Name:lower()
    
    if weaponName:find("kiem") or weaponName:find("sword") or weaponName:find("blade") then
        -- Kiếm: dùng handle đánh
        local handle = weapon:FindFirstChild("Handle")
        if handle then
            handle.CFrame = enemy.HumanoidRootPart.CFrame
        end
        weapon:Activate()
        task.wait(0.05)
        weapon:Deactivate()
        return true
    elseif weaponName:find("nam") or weaponName:find("fist") or weaponName:find("punch") or weaponName:find("combat") then
        -- Nấm đấm: dùng remote đánh
        local remotes = {"MeleeAttack", "Attack", "Punch", "Hit", "Click"}
        for _, remoteName in pairs(remotes) do
            local remote = character:FindFirstChild(remoteName) or weapon:FindFirstChild(remoteName)
            if remote and remote:IsA("RemoteEvent") then
                remote:FireServer(enemy)
                return true
            end
        end
        
        weapon:Activate()
        task.wait(0.05)
        weapon:Deactivate()
        return true
    else
        -- Vũ khí khác
        weapon:Activate()
        task.wait(0.05)
        weapon:Deactivate()
        return true
    end
end

-- Auto Quest
function StartAutoQuest()
    if Connections.AutoQuest then return end
    
    Connections.AutoQuest = RunService.RenderStepped:Connect(function()
        if not Settings.AutoQuest then return end
        
        local character = LocalPlayer.Character
        if not character or not character:FindFirstChild("HumanoidRootPart") then return end
        
        local rootPart = character.HumanoidRootPart
        
        for _, obj in pairs(workspace:GetDescendants()) do
            if obj:IsA("ProximityPrompt") then
                local npc = obj.Parent
                if npc and npc:IsA("Model") and npc:FindFirstChild("HumanoidRootPart") then
                    local distance = (rootPart.Position - npc.HumanoidRootPart.Position).Magnitude
                    
                    if distance < 100 then
                        rootPart.CFrame = npc.HumanoidRootPart.CFrame + Vector3.new(0, 3, 2)
                        fireproximityprompt(obj)
                        
                        for _, remote in pairs(ReplicatedStorage:GetDescendants()) do
                            if remote:IsA("RemoteEvent") then
                                local name = remote.Name:lower()
                                if name:find("quest") or name:find("accept") then
                                    remote:FireServer("Accept")
                                    break
                                end
                            end
                        end
                        
                        break
                    end
                end
            end
        end
    end)
end

function StopAutoQuest()
    if Connections.AutoQuest then
        Connections.AutoQuest:Disconnect()
        Connections.AutoQuest = nil
    end
end

-- Auto Farm - Dịch chuyển đến quái
function StartAutoFarm()
    if Connections.AutoFarm then return end
    
    Connections.AutoFarm = RunService.RenderStepped:Connect(function()
        if not Settings.AutoFarm then return end
        
        local character = LocalPlayer.Character
        if not character or not character:FindFirstChild("HumanoidRootPart") then return end
        
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        if not humanoid or humanoid.Health <= 0 then return end
        
        local rootPart = character.HumanoidRootPart
        local enemy = FindNearestEnemy(Settings.AttackRange)
        
        if enemy then
            rootPart.CFrame = CFrame.new(enemy.HumanoidRootPart.Position + Vector3.new(0, 3, 2))
        end
    end)
end

function StopAutoFarm()
    if Connections.AutoFarm then
        Connections.AutoFarm:Disconnect()
        Connections.AutoFarm = nil
    end
end

-- Auto Damage - Đánh bằng vũ khí, mỗi 1 giây
function StartAutoDamage()
    if Connections.AutoDamage then return end
    
    Connections.AutoDamage = RunService.RenderStepped:Connect(function()
        if not Settings.AutoDamage then return end
        
        local character = LocalPlayer.Character
        if not character or not character:FindFirstChild("HumanoidRootPart") then return end
        
        local rootPart = character.HumanoidRootPart
        
        -- Kiểm tra có vũ khí không
        local weapon = GetEquippedWeapon() or GetBackpackWeapon()
        if not weapon then
            Notify("BIN BELL", "Chua co vu khi! Trang bi nam dam hoac kiem truoc")
            return
        end
        
        -- Tìm tất cả quái trong phạm vi
        local allEnemies = FindEnemies()
        local attacked = false
        
        for _, enemy in pairs(allEnemies) do
            if not IsPlayer(enemy) then
                local distance = (rootPart.Position - enemy.HumanoidRootPart.Position).Magnitude
                
                if distance <= Settings.DamageRange then
                    -- Dịch chuyển đến quái để đánh
                    rootPart.CFrame = CFrame.new(enemy.HumanoidRootPart.Position + Vector3.new(0, 3, 2))
                    
                    -- Đánh bằng vũ khí
                    AttackWithWeapon(enemy)
                    attacked = true
                    break
                end
            end
        end
        
        -- Đánh người chơi đối địch
        if not attacked then
            for _, player in pairs(Players:GetPlayers()) do
                if player ~= LocalPlayer and player.Character then
                    local char = player.Character
                    if char:FindFirstChild("Humanoid") and char:FindFirstChild("HumanoidRootPart") then
                        if char.Humanoid.Health > 0 then
                            local isEnemy = true
                            if LocalPlayer.Team and player.Team then
                                if LocalPlayer.Team == player.Team then
                                    isEnemy = false
                                end
                            end
                            
                            if isEnemy then
                                local distance = (rootPart.Position - char.HumanoidRootPart.Position).Magnitude
                                if distance <= Settings.DamageRange then
                                    rootPart.CFrame = CFrame.new(char.HumanoidRootPart.Position + Vector3.new(0, 3, 2))
                                    AttackWithWeapon(char)
                                    break
                                end
                            end
                        end
                    end
                end
            end
        end
        
        -- Chờ 1 giây
        task.wait(Settings.DamageInterval)
    end)
end

function StopAutoDamage()
    if Connections.AutoDamage then
        Connections.AutoDamage:Disconnect()
        Connections.AutoDamage = nil
    end
end

-- Fix Lag
function StartFixLag()
    Lighting.GlobalShadows = false
    Lighting.Brightness = 1.5
    Lighting.FogEnd = 100
    Lighting.FogStart = 50
    
    if Connections.FixLag then return end
    
    Connections.FixLag = RunService.RenderStepped:Connect(function()
        if not Settings.FixLag then return end
        
        for _, obj in pairs(workspace:GetDescendants()) do
            if obj:IsA("Decal") or obj:IsA("ParticleEmitter") or obj:IsA("Smoke") or obj:IsA("Fire") or obj:IsA("Texture") then
                obj:Destroy()
            end
        end
    end)
end

function StopFixLag()
    Lighting.GlobalShadows = true
    Lighting.Brightness = 1
    Lighting.FogEnd = 10000
    Lighting.FogStart = 5000
    
    if Connections.FixLag then
        Connections.FixLag:Disconnect()
        Connections.FixLag = nil
    end
end

-- ESP
function AddHighlight(target, color)
    if not target or not target:IsA("Model") then return end
    if target:FindFirstChild("BinBellHighlight") then return end
    
    local highlight = Instance.new("Highlight")
    highlight.Name = "BinBellHighlight"
    highlight.FillColor = color
    highlight.OutlineColor = color
    highlight.FillTransparency = 0.5
    highlight.OutlineTransparency = 0
    highlight.Parent = target
    
    table.insert(ESPHighlights, highlight)
end

function ClearHighlights()
    for _, h in pairs(ESPHighlights) do
        if h and h.Parent then
            h:Destroy()
        end
    end
    ESPHighlights = {}
end

function StartESP()
    if Connections.ESP then return end
    
    Connections.ESP = RunService.RenderStepped:Connect(function()
        if Settings.ESPEnemies then
            local allEnemies = FindEnemies()
            for _, enemy in pairs(allEnemies) do
                if not enemy:FindFirstChild("BinBellHighlight") then
                    AddHighlight(enemy, Color3.fromRGB(255, 0, 0))
                end
            end
        end
        
        if Settings.ESPPlayers then
            for _, player in pairs(Players:GetPlayers()) do
                if player ~= LocalPlayer and player.Character then
                    local char = player.Character
                    if char:FindFirstChild("Humanoid") and char:FindFirstChild("HumanoidRootPart") then
                        if char.Humanoid.Health > 0 then
                            if not char:FindFirstChild("BinBellHighlight") then
                                AddHighlight(char, Color3.fromRGB(0, 255, 255))
                            end
                        end
                    end
                end
            end
        end
    end)
end

function StopESP()
    ClearHighlights()
    if Connections.ESP then
        Connections.ESP:Disconnect()
        Connections.ESP = nil
    end
end
