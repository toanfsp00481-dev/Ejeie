local ESP = {
    Enabled = false,
    PlayerESPEnabled = false,
    EnemyESPEnabled = false,
    Boxes = {},
    Labels = {},
    Connection = nil
}

function ESP:CreateBox(target, color)
    -- Tạo box 3D chiếu lên màn hình
    local box = Drawing.new("Square")
    box.Thickness = 2
    box.Color = color
    box.Filled = false
    box.Transparency = 0.7
    box.Visible = false
    return box
end

function ESP:CreateLabel(target, color)
    -- Tạo nhãn tên
    local label = Drawing.new("Text")
    label.Size = 13
    label.Color = color
    label.Center = true
    label.Outline = true
    label.OutlineColor = Color3.fromRGB(0, 0, 0)
    label.Visible = false
    return label
end

function ESP:GetTeamColor(player)
    -- Trả về màu theo team
    if player.Team then
        return player.Team.TeamColor.Color
    end
    return Color3.fromRGB(255, 255, 255)
end

function ESP:Update()
    if not self.Enabled then return end
    
    -- Xóa box cũ không còn dùng
    for target, data in pairs(self.Boxes) do
        if not target or not target.Parent then
            if data.Box then data.Box:Remove() end
            if data.Label then data.Label:Remove() end
            self.Boxes[target] = nil
            self.Labels[target] = nil
        end
    end
    
    -- ESP cho người chơi
    if self.PlayerESPEnabled then
        for _, player in pairs(Players:GetPlayers()) do
            if player ~= LocalPlayer and player.Character then
                local character = player.Character
                local rootPart = character:FindFirstChild("HumanoidRootPart")
                local humanoid = character:FindFirstChildOfClass("Humanoid")
                
                if rootPart and humanoid and humanoid.Health > 0 then
                    local screenPos, onScreen = Camera:WorldToViewportPoint(rootPart.Position)
                    
                    if onScreen then
                        -- Tạo hoặc cập nhật box
                        if not self.Boxes[player] then
                            local color = self:GetTeamColor(player)
                            self.Boxes[player] = {
                                Box = self:CreateBox(player, color),
                                Label = self:CreateLabel(player, color)
                            }
                        end
                        
                        local boxData = self.Boxes[player]
                        local box = boxData.Box
                        local label = boxData.Label
                        
                        -- Tính toán kích thước box dựa trên khoảng cách
                        local distance = (Camera.CFrame.Position - rootPart.Position).Magnitude
                        local scale = math.clamp(2000 / distance, 20, 80)
                        
                        -- Vẽ box
                        box.Visible = true
                        box.Position = Vector2.new(screenPos.X - scale/2, screenPos.Y - scale/2)
                        box.Size = Vector2.new(scale, scale * 1.5)
                        
                        -- Vẽ label
                        label.Visible = true
                        label.Position = Vector2.new(screenPos.X, screenPos.Y - scale/2 - 20)
                        label.Text = player.Name .. " [" .. math.floor(humanoid.Health) .. " HP]"
                    else
                        if self.Boxes[player] then
                            self.Boxes[player].Box.Visible = false
                            self.Boxes[player].Label.Visible = false
                        end
                    end
                end
            end
        end
    end
    
    -- ESP cho quái vật
    if self.EnemyESPEnabled then
        for _, enemy in pairs(Workspace.Enemies:GetChildren()) do
            if enemy:FindFirstChild("HumanoidRootPart") and enemy:FindFirstChild("Humanoid") then
                local rootPart = enemy.HumanoidRootPart
                local humanoid = enemy.Humanoid
                
                if humanoid.Health > 0 then
                    local screenPos, onScreen = Camera:WorldToViewportPoint(rootPart.Position)
                    
                    if onScreen then
                        if not self.Boxes[enemy] then
                            local color = Color3.fromRGB(255, 0, 0)
                            self.Boxes[enemy] = {
                                Box = self:CreateBox(enemy, color),
                                Label = self:CreateLabel(enemy, color)
                            }
                        end
                        
                        local boxData = self.Boxes[enemy]
                        local box = boxData.Box
                        local label = boxData.Label
                        
                        local distance = (Camera.CFrame.Position - rootPart.Position).Magnitude
                        local scale = math.clamp(1800 / distance, 15, 60)
                        
                        box.Visible = true
                        box.Position = Vector2.new(screenPos.X - scale/2, screenPos.Y - scale/2)
                        box.Size = Vector2.new(scale, scale * 1.3)
                        
                        label.Visible = true
                        label.Position = Vector2.new(screenPos.X, screenPos.Y - scale/2 - 18)
                        label.Text = enemy.Name .. " [" .. math.floor(humanoid.Health) .. " HP]"
                    else
                        if self.Boxes[enemy] then
                            self.Boxes[enemy].Box.Visible = false
                            self.Boxes[enemy].Label.Visible = false
                        end
                    end
                end
            end
        end
    end
end

function ESP:Start()
    self.Enabled = true
    
    if not self.Connection then
        self.Connection = RunService.RenderStepped:Connect(function()
            self:Update()
        end)
    end
end

function ESP:Stop()
    self.Enabled = false
    
    -- Ẩn tất cả box
    for target, data in pairs(self.Boxes) do
        if data.Box then data.Box:Remove() end
        if data.Label then data.Label:Remove() end
    end
    self.Boxes = {}
    self.Labels = {}
    
    if self.Connection then
        self.Connection:Disconnect()
        self.Connection = nil
    end
end

-- HỆ THỐNG AUTO FARM (Nâng cấp siêu thanh)
-- ============================================
local AutoFarm = {
    Enabled = false,
    CurrentTarget = nil,
    AttackRange = 15,
    LastAttack = 0,
    AttackDelay = 0.3,
    SuperAttackEnabled = false
}

function AutoFarm:Start()
    self.Enabled = true
    
    self.Connection = RunService.RenderStepped:Connect(function()
        if not self.Enabled then return end
        
        local character = LocalPlayer.Character
        if not character or not character:FindFirstChild("HumanoidRootPart") then return end
        
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        if not humanoid or humanoid.Health <= 0 then return end
        
        -- Tìm enemy gần nhất
        local nearestEnemy = nil
        local nearestDistance = self.AttackRange
        
        for _, enemy in pairs(Workspace.Enemies:GetChildren()) do
            if enemy:FindFirstChild("HumanoidRootPart") and enemy:FindFirstChild("Humanoid") then
                local enemyHumanoid = enemy.Humanoid
                if enemyHumanoid.Health > 0 then
                    local distance = (character.HumanoidRootPart.Position - enemy.HumanoidRootPart.Position).Magnitude
                    if distance < nearestDistance then
                        nearestDistance = distance
                        nearestEnemy = enemy
                    end
                end
            end
        end
        
        if nearestEnemy then
            self.CurrentTarget = nearestEnemy
            
            -- Dịch chuyển đến enemy
            local targetPos = nearestEnemy.HumanoidRootPart.Position
            character.HumanoidRootPart.CFrame = CFrame.new(targetPos + Vector3.new(0, 3, 0))
            
            -- Kiểm tra delay tấn công
            local currentDelay = self.SuperAttackEnabled and 0.01 or self.AttackDelay
            
            if os.clock() - self.LastAttack >= currentDelay then
                self.LastAttack = os.clock()
                
                -- Tấn công với nhiều remote khác nhau
                local args = {
                    [1] = "Attack",
                    [2] = nearestEnemy
                }
                
                for _, remoteName in pairs({"MeleeAttack", "Attack", "Click", "Hit", "Swing"}) do
                    local remote = LocalPlayer.Backpack:FindFirstChild(remoteName) or 
                                   character:FindFirstChild(remoteName)
                    if remote and remote:IsA("RemoteEvent") then
                        remote:FireServer(unpack(args))
                        break
                    end
                end
                
                -- Thử thêm các tool tấn công
                for _, tool in pairs(character:GetChildren()) do
                    if tool:IsA("Tool") and tool:FindFirstChild("Handle") then
                        local handle = tool.Handle
                        handle.CFrame = nearestEnemy.HumanoidRootPart.CFrame
                        tool:Activate()
                    end
                end
            end
        end
    end)
end

function AutoFarm:Stop()
    self.Enabled = false
    if self.Connection then
        self.Connection:Disconnect()
        self.Connection = nil
    end
end

-- ============================================
-- HỆ THỐNG CHỐNG AFK
-- ============================================
local AntiAFK = {
    Enabled = false,
    Connection = nil
}

function AntiAFK:Start()
    self.Enabled = true
    
    self.Connection = RunService.RenderStepped:Connect(function()
        if not self.Enabled then return end
        
        -- Gửi input giả
        VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.LeftShift, false, nil)
        VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.LeftShift, false, nil)
    end)
end

function AntiAFK:Stop()
    self.Enabled = false
    if self.Connection then
        self.Connection:Disconnect()
        self.Connection = nil
    end
end
