-- HỆ THỐNG ESP DÙNG HIGHLIGHT
-- ============================================
local function AddHighlight(target, color)
    if not target or not target:IsA("Model") then return end
    if target:FindFirstChild("BinBellHighlight") then return end
    
    local highlight = Instance.new("Highlight")
    highlight.Name = "BinBellHighlight"
    highlight.FillColor = color
    highlight.OutlineColor = color
    highlight.FillTransparency = 0.5
    highlight.OutlineTransparency = 0
    highlight.Enabled = true
    highlight.Parent = target
    
    table.insert(ESPHighlights, highlight)
end

local function RemoveHighlight(target)
    if not target then return end
    
    local highlight = target:FindFirstChild("BinBellHighlight")
    if highlight then
        highlight:Destroy()
    end
end

local function ClearAllHighlights()
    for _, highlight in pairs(ESPHighlights) do
        if highlight and highlight.Parent then
            highlight:Destroy()
        end
    end
    ESPHighlights = {}
end

local function StartESP()
    if ESPConnection then return end
    
    ESPConnection = RunService.RenderStepped:Connect(function()
        -- ESP Quái Vật
        if Settings.ESPEnemies then
            for _, enemy in pairs(workspace.Enemies:GetChildren()) do
                if enemy:FindFirstChild("HumanoidRootPart") and enemy:FindFirstChild("Humanoid") then
                    if enemy.Humanoid.Health > 0 then
                        if not enemy:FindFirstChild("BinBellHighlight") then
                            AddHighlight(enemy, Color3.fromRGB(255, 0, 0))
                        end
                    else
                        RemoveHighlight(enemy)
                    end
                end
            end
        end
        
        -- ESP Người Chơi
        if Settings.ESPPlayers then
            for _, player in pairs(Players:GetPlayers()) do
                if player ~= LocalPlayer and player.Character then
                    local character = player.Character
                    if character:FindFirstChild("HumanoidRootPart") and character:FindFirstChild("Humanoid") then
                        if character.Humanoid.Health > 0 then
                            if not character:FindFirstChild("BinBellHighlight") then
                                local color = Color3.fromRGB(0, 255, 255)
                                if player.Team then
                                    color = player.Team.TeamColor.Color
                                end
                                AddHighlight(character, color)
                            end
                        else
                            RemoveHighlight(character)
                        end
                    end
                end
            end
        end
    end)
end

local function StopESP()
    ClearAllHighlights()
    
    if ESPConnection then
        ESPConnection:Disconnect()
        ESPConnection = nil
    end
end

-- HỆ THỐNG AUTO FARM
-- ============================================
local function StartAutoFarm()
    if AutoFarmConnection then return end
    
    AutoFarmConnection = RunService.RenderStepped:Connect(function()
        if not Settings.AutoFarm then return end
        
        local character = LocalPlayer.Character
        if not character or not character:FindFirstChild("HumanoidRootPart") then return end
        
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        if not humanoid or humanoid.Health <= 0 then return end
        
        -- Tìm quái gần nhất
        local nearestEnemy = nil
        local nearestDistance = 20
        
        for _, enemy in pairs(workspace.Enemies:GetChildren()) do
            if enemy:FindFirstChild("HumanoidRootPart") and enemy:FindFirstChild("Humanoid") then
                if enemy.Humanoid.Health > 0 then
                    local distance = (character.HumanoidRootPart.Position - enemy.HumanoidRootPart.Position).Magnitude
                    if distance < nearestDistance then
                        nearestDistance = distance
                        nearestEnemy = enemy
                    end
                end
            end
        end
        
        if nearestEnemy then
            -- Dịch chuyển đến quái
            character.HumanoidRootPart.CFrame = nearestEnemy.HumanoidRootPart.CFrame + Vector3.new(0, 3, 0)
            
            -- Tấn công
            local currentTime = os.clock()
            local attackDelay = Settings.SuperAttack and 0.01 or 0.3
            
            -- Kích hoạt tool
            for _, tool in pairs(character:GetChildren()) do
                if tool:IsA("Tool") and tool:FindFirstChild("Handle") then
                    tool:Activate()
                end
            end
            
            -- Thử fire remote tấn công
            local remotes = {"MeleeAttack", "Attack", "Click", "Hit", "Swing", "Punch"}
            for _, remoteName in pairs(remotes) do
                local remote = character:FindFirstChild(remoteName) or LocalPlayer.Backpack:FindFirstChild(remoteName)
                if remote and remote:IsA("RemoteEvent") then
                    remote:FireServer("Attack", nearestEnemy)
                    break
                end
            end
            
            -- Kích hoạt tool trong backpack
            for _, tool in pairs(LocalPlayer.Backpack:GetChildren()) do
                if tool:IsA("Tool") and tool:FindFirstChild("Handle") then
                    tool:Activate()
                end
            end
        end
    end)
end

local function StopAutoFarm()
    if AutoFarmConnection then
        AutoFarmConnection:Disconnect()
        AutoFarmConnection = nil
    end
end

local function StartAntiAFK()
    if AntiAFKConnection then return end
    
    AntiAFKConnection = RunService.RenderStepped:Connect(function()
        if not Settings.AntiAFK then return end
        
        local VirtualInputManager = game:GetService("VirtualInputManager")
        VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.LeftShift, false, nil)
        VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.LeftShift, false, nil)
    end)
end

local function StopAntiAFK()
    if AntiAFKConnection then
        AntiAFKConnection:Disconnect()
        AntiAFKConnection = nil
    end
end

function UI:ToggleMenu()
    self.MenuOpen = not self.MenuOpen
    
    if self.MenuOpen then
        self.MainFrame.Visible = true
        local tween = TweenService:Create(self.MainFrame,
            TweenInfo.new(0.3, Enum.EasingStyle.Quint, Enum.EasingDirection.Out),
            {Position = UDim2.new(0, 10, 0, 10)}
        )
        tween:Play()
    else
        local tween = TweenService:Create(self.MainFrame,
            TweenInfo.new(0.3, Enum.EasingStyle.Quint, Enum.EasingDirection.In),
            {Position = UDim2.new(0, 10, 0, -430)}
        )
        tween:Play()
        tween.Completed:Connect(function()
            self.MainFrame.Visible = false
        end)
    end
end
