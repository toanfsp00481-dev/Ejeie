-- ========== HELPERS ==========
local camPosCache = Vector3.zero
local function UpdateCamCache()
    camPosCache = Camera.CFrame.Position
end

local function GetAliveChar(player)
    local char = player.Character
    if not char then return nil end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return nil end
    return char
end

local function IsValidTarget(player, char)
    if player == LocalPlayer then return false end
    if not char then return false end
    if Config.TeamCheck and player.Team and LocalPlayer.Team and player.Team == LocalPlayer.Team then
        return false
    end
    return true
end

-- ========== TARGET CACHE ==========
local headCache = { head = nil, dist = math.huge }
local bodyCache = { part = nil, dist = math.huge }
local centerCache = { head = nil, dist = math.huge }
local lastTargetUpdate = 0
local lastCenterUpdate = 0

local function RefreshTargets(now)
    if now - lastTargetUpdate < Config.TargetUpdate then return end
    lastTargetUpdate = now

    local bestHead, bestHeadDist = nil, Config.SilentFOV
    local bestBody, bestBodyDist = nil, Config.AimBodyFOV
    local mx, my = Mouse.X, Mouse.Y
    local camPos = camPosCache

    for _, player in pairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            local char = player.Character
            if char and IsValidTarget(player, char) then
                local hum = char:FindFirstChildOfClass("Humanoid")
                if hum and hum.Health > 0 then
                    local head = char:FindFirstChild("Head")
                    local bodyPart = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso")
                    
                    if head and (head.Position - camPos).Magnitude < 500 then
                        local sp, onScreen = Camera:WorldToViewportPoint(head.Position)
                        if onScreen then
                            local d = (Vector2.new(sp.X, sp.Y) - Vector2.new(mx, my)).Magnitude
                            if d < bestHeadDist then
                                bestHeadDist = d
                                bestHead = head
                            end
                        end
                    end
                    
                    if bodyPart and (bodyPart.Position - camPos).Magnitude < 500 then
                        local sp, onScreen = Camera:WorldToViewportPoint(bodyPart.Position)
                        if onScreen then
                            local d = (Vector2.new(sp.X, sp.Y) - Vector2.new(mx, my)).Magnitude
                            if d < bestBodyDist then
                                bestBodyDist = d
                                bestBody = bodyPart
                            end
                        end
                    end
                end
            end
        end
    end
    headCache.head = bestHead
    headCache.dist = bestHeadDist
    bodyCache.part = bestBody
    bodyCache.dist = bestBodyDist
end

local function RefreshCenter(now)
    if now - lastCenterUpdate < Config.TargetUpdate then return end
    lastCenterUpdate = now

    local best, bestDist = nil, Config.CenterFOV
    local vp = Camera.ViewportSize
    local cx, cy = vp.X / 2, vp.Y / 2
    local camPos = camPosCache

    for _, player in pairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            local char = player.Character
            if char and IsValidTarget(player, char) then
                local hum = char:FindFirstChildOfClass("Humanoid")
                if hum and hum.Health > 0 then
                    local head = char:FindFirstChild("Head")
                    if head and (head.Position - camPos).Magnitude < 500 then
                        local sp, onScreen = Camera:WorldToViewportPoint(head.Position)
                        if onScreen then
                            local d = (Vector2.new(sp.X, sp.Y) - Vector2.new(cx, cy)).Magnitude
                            if d < bestDist then
                                bestDist = d
                                best = head
                            end
                        end
                    end
                end
            end
        end
    end
    centerCache.head = best
    centerCache.dist = bestDist
end

-- ========== NOCLIP ==========
local noclipRunning = false
function ApplyNoclip()
    if noclipRunning then return end
    noclipRunning = true
    task.spawn(function()
        while Config.Noclip do
            local char = LocalPlayer.Character
            if char then
                for _, part in ipairs(char:GetDescendants()) do
                    if part:IsA("BasePart") and part.CanCollide then
                        part.CanCollide = false
                    end
                end
            end
            task.wait(0.1)
        end
        noclipRunning = false
    end)
end

-- ========== AIM BODY (hitbox expander) ==========
local aimBodyConn = nil
local originalSizes = {}

function ApplyAimBody()
    RemoveAimBody()
    aimBodyConn = RunService.Heartbeat:Connect(Safe(function()
        if not Config.AimBody then return end
        for _, p in pairs(Players:GetPlayers()) do
            if p ~= LocalPlayer then
                local char = p.Character
                if char and IsValidTarget(p, char) then
                    local hum = char:FindFirstChildOfClass("Humanoid")
                    if hum and hum.Health > 0 then
                        for _, part in ipairs(char:GetDescendants()) do
                            if part:IsA("BasePart") and part.Name ~= "Head" then
                                if not originalSizes[part] then
                                    originalSizes[part] = part.Size
                                end
                                local orig = originalSizes[part]
                                local newSize = Vector3.new(
                                    math.max(orig.X, 3),
                                    math.max(orig.Y, 3),
                                    math.max(orig.Z, 3)
                                )
                                if part.Size ~= newSize then
                                    part.Size = newSize
                                end
                            end
                        end
                    end
                end
            end
        end
    end))
end

function RemoveAimBody()
    if aimBodyConn then
        pcall(function() aimBodyConn:Disconnect() end)
        aimBodyConn = nil
    end
    for part, size in pairs(originalSizes) do
        if part and part.Parent then
            pcall(function() part.Size = size end)
        end
    end
    originalSizes = {}
end

-- ========== ESP ==========
local espCache = {}
local lastEspUpdate = 0

local function UpdateESP(now)
    if now - lastEspUpdate < Config.ESPRate then return end
    lastEspUpdate = now

    for _, player in pairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            local char = GetAliveChar(player)
            if Config.ESP and char then
                local existing = espCache[player]
                if not existing or existing.Parent ~= char then
                    if existing then pcall(function() existing:Destroy() end) end
                    local h = Instance.new("Highlight")
                    h.Name = "CyberESP"
                    h.FillColor = Theme.Accent
                    h.FillTransparency = 0.6
                    h.OutlineColor = Theme.Accent2
                    h.OutlineTransparency = 0
                    h.Parent = char
                    espCache[player] = h
                end
            elseif not Config.ESP and espCache[player] then
                pcall(function() espCache[player]:Destroy() end)
                espCache[player] = nil
            end
        end
    end
end

local silentAimHooked = false
local silentAimMethod = "none"
local silentActive = false
local cachedHitCFrame = nil
local cachedTargetPart = nil

local function SetupSilentAim()
    if silentAimHooked then return end

    local hasGetRawMeta = (typeof(getrawmetatable) == "function")
    local hasSetReadonly = (typeof(setreadonly) == "function")
    local hasIsReadonly  = (typeof(isreadonly) == "function")

    if hasGetRawMeta then
        local okA, errA = pcall(function()
            local mt = getrawmetatable(Mouse)
            if not mt then error("no mouse metatable") end
            local oldIndex = rawget(mt, "__index")
            if not oldIndex then error("no __index") end

            local wasRO = false
            if hasIsReadonly and hasSetReadonly then
                wasRO = isreadonly(mt)
                if wasRO then setreadonly(mt, false) end
            end

            rawset(mt, "__index", function(self, key)
                if not silentActive then
                    return oldIndex(self, key)
                end
                if key == "Hit" then
                    return cachedHitCFrame or oldIndex(self, key)
                elseif key == "Target" then
                    return cachedTargetPart or oldIndex(self, key)
                end
                return oldIndex(self, key)
            end)

            if wasRO and hasSetReadonly then setreadonly(mt, true) end
        end)

        if okA then
            silentAimHooked = true
            silentAimMethod = "getrawmetatable(Mouse) ✓"
            print("⚡ Silent Aim: " .. silentAimMethod)
            return
        end
    end

    if typeof(hookmetamethod) == "function" then
        local okB = pcall(function()
            local oldIndex
            oldIndex = hookmetamethod(game, "__index", function(self, key)
                if not silentActive then return oldIndex(self, key) end
                if key ~= "Hit" and key ~= "Target" then return oldIndex(self, key) end
                if typeof(self) == "Instance" and self.ClassName == "Mouse" then
                    if key == "Hit" then
                        return cachedHitCFrame or oldIndex(self, key)
                    elseif key == "Target" then
                        return cachedTargetPart or oldIndex(self, key)
                    end
                end
                return oldIndex(self, key)
            end)
        end)
        if okB then
            silentAimHooked = true
            silentAimMethod = "__index fallback"
            print("⚡ Silent Aim: " .. silentAimMethod)
            return
        end
    end

    Config.SilentAim = false
    warn("⚠️ Silent Aim không khả dụng trên executor này")
end

pcall(SetupSilentAim)

-- Update silent cache (main loop)
local function UpdateSilentCache()
    if not Config.SilentAim then
        silentActive = false
        cachedHitCFrame = nil
        cachedTargetPart = nil
        return
    end

    -- Silent Aim ưu tiên head, nếu CenterLock bật thì dùng centerCache
    local target = Config.CenterLock and centerCache.head or headCache.head
    if target and target.Parent and target:IsDescendantOf(Workspace) then
        local ok, cf = pcall(function()
            return CFrame.new(camPosCache, target.Position)
        end)
        if ok and cf then
            cachedHitCFrame = cf
            cachedTargetPart = target
            silentActive = true
            return
        end
    end
    silentActive = false
    cachedHitCFrame = nil
    cachedTargetPart = nil
end
